CONTEXTO DE NEGOCIO: fuera de esta app corre un cron en n8n que, todos los

### 3.3 — Endpoint para que n8n reporte el resultado

Crear `app/api/notificaciones/webhook/route.ts`:

```ts
import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";

export async function POST(request: Request) {
  const secreto = request.headers.get("x-n8n-secret");
  if (!secreto || secreto !== process.env.N8N_WEBHOOK_SECRET) {
    return NextResponse.json({ error: "No autorizado." }, { status: 401 });
  }

  const datos = await request.json();
  if (!datos.suscripcionId || !datos.estado) {
    return NextResponse.json({ error: "Faltan campos obligatorios." }, { status: 400 });
  }
  if (datos.estado !== "enviado" && datos.estado !== "fallido") {
    return NextResponse.json({ error: "estado debe ser 'enviado' o 'fallido'." }, { status: 400 });
  }

  const suscripcion = await prisma.suscripcion.findUnique({
    where: { id: datos.suscripcionId },
  });
  if (!suscripcion) {
    return NextResponse.json({ error: "La suscripción indicada no existe." }, { status: 404 });
  }

  const notificacion = await prisma.notificacion.create({
    data: {
      suscripcionId: suscripcion.id,
      clienteId: suscripcion.clienteId,
      canal: datos.canal || "whatsapp",
      estado: datos.estado,
      detalle: datos.detalle || null,
    },
  });
  return NextResponse.json(notificacion, { status: 201 });
}
```

Notar que `clienteId` se resuelve de `suscripcion.clienteId` (ya buscado en
la base), nunca del body de la request — así ningún llamador puede
falsificar de qué cliente es la notificación aunque tenga el secreto.

Cómo verificar (con `npm run dev` corriendo, sin sesión de browser — este
endpoint no usa cookies):
```bash

curl -X POST http://localhost:3000/api/notificaciones/webhook \
  -H "Content-Type: application/json" \
  -d '{"suscripcionId":"x","estado":"enviado"}'

# Con secreto incorrecto -> 401
curl -X POST http://localhost:3000/api/notificaciones/webhook \
  -H "Content-Type: application/json" -H "x-n8n-secret: incorrecto" \

# Con secreto correcto pero suscripcionId inventado -> 404
curl -X POST http://localhost:3000/api/notificaciones/webhook \

```

import { NextResponse } from "next/server";
import { filtroClientePorSesion, requireSesion } from "@/lib/auth";
import { prisma } from "@/lib/prisma";

export async function GET() {
  const sesion = await requireSesion();
  if (!sesion) {
    return NextResponse.json({ error: "No autenticado." }, { status: 401 });
  }

  const notificaciones = await prisma.notificacion.findMany({
    where: { cliente: filtroClientePorSesion(sesion) },
    include: {
      cliente: { select: { nombre: true } },
      suscripcion: { select: { servicio: true } },
    },
    orderBy: { enviadoEn: "desc" },
  });
  return NextResponse.json(notificaciones);
}

### 3.5 — Hook de datos y pantalla del dashboard

Crear `lib/notificaciones-store.ts`:

```ts
"use client";

import { useEffect, useState } from "react";

export type Notificacion = {
  id: string;
  suscripcionId: string;
  clienteId: string;
  canal: string;
  estado: string;
  detalle: string | null;
  enviadoEn: string;
  cliente: { nombre: string };
  suscripcion: { servicio: string };
};

export function useNotificaciones() {
  const [notificaciones, setNotificaciones] = useState<Notificacion[]>([]);
  const [cargando, setCargando] = useState(true);

  useEffect(() => {
    fetch("/api/notificaciones")
      .then((res) => (res.ok ? res.json() : []))
      .then((datos: Notificacion[]) => setNotificaciones(datos))
      .finally(() => setCargando(false));
  }, []);

  return { notificaciones, cargando };
}
```

export default function NotificacionesPage() {
  const { notificaciones, cargando } = useNotificaciones();

  if (cargando) {
    return (
      <div className="flex flex-col gap-6 px-6 py-8 sm:px-10">
        <p className="text-sm text-[var(--color-text-muted)]">Cargando...</p>
      </div>
    );
  }

  return (
    <div className="flex flex-col gap-6 px-6 py-8 sm:px-10">
      <h1 className="text-xl font-medium text-[var(--color-text)]">Notificaciones</h1>

      <div className="overflow-x-auto">
        <table className="w-full min-w-[640px] text-left text-sm">
          <thead>
            <tr className="border-b border-[var(--color-border)] text-[var(--color-text-muted)]">
              <th className="py-2 pr-4 font-medium">Cliente</th>
              <th className="py-2 pr-4 font-medium">Servicio</th>
              <th className="py-2 pr-4 font-medium">Canal</th>
              <th className="py-2 pr-4 font-medium">Estado</th>
              <th className="py-2 pr-4 font-medium">Fecha</th>
              <th className="py-2 pr-4 font-medium">Detalle</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-[var(--color-border)]">
            {notificaciones.map((n) => (
              <tr key={n.id}>
                <td className="py-3 pr-4 text-[var(--color-text)]">{n.cliente.nombre}</td>
                <td className="py-3 pr-4 text-[var(--color-text-muted)]">{n.suscripcion.servicio}</td>
                <td className="py-3 pr-4 text-[var(--color-text-muted)]">{n.canal}</td>
                <td className="py-3 pr-4">
                  <span
                    className={
                      n.estado === "enviado"
                        ? "text-[var(--color-success-text)]"
                        : "text-[var(--color-danger-text)]"
                    }
                  >
                    {n.estado === "enviado" ? "Enviado" : "Falló"}
                  </span>
                </td>
                <td className="py-3 pr-4 text-[var(--color-text-muted)]">{formatFecha(n.enviadoEn)}</td>
                <td className="py-3 pr-4 text-[var(--color-text-muted)]">{n.detalle ?? "—"}</td>
              </tr>
            ))}
            {notificaciones.length === 0 && (
              <tr>
                <td colSpan={6} className="py-6 text-center text-[var(--color-text-muted)]">
                  Todavía no hay notificaciones registradas.
                </td>
              </tr>
            )}
          </tbody>
        </table>
      </div>
    </div>
  );
}
